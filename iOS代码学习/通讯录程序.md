## DBManager.m

```objc
#import "DBManager.h"
#import <FMDB/FMDB.h>

@implementation DBManager {
    FMDatabaseQueue *_queue;
}

+ (instancetype)sharedManager {
    static DBManager *_instance = nil;
    static dispatch_once_t onceToken;
    dispatch_once(&onceToken, ^{
        _instance = [[DBManager alloc] init];
    });
    return _instance;
}

- (instancetype)init {
    self = [super init];
    if (self) {
        NSString *docPath = [NSSearchPathForDirectoriesInDomains(NSDocumentDirectory, NSUserDomainMask, YES) firstObject];
        NSString *dbPath = [docPath stringByAppendingPathComponent:@"contact.db"];
        _queue = [FMDatabaseQueue databaseQueueWithPath:dbPath];
        [self createTable];
    }
    return self;
}

- (void)createTable {
    [_queue inDatabase:^(FMDatabase *db) {
        NSString *sql = @"CREATE TABLE IF NOT EXISTS contact ("
                         "id INTEGER PRIMARY KEY AUTOINCREMENT,"
                         "name TEXT NOT NULL,"
                         "phone TEXT NOT NULL,"
                         "avatar TEXT)";
        [db executeUpdate:sql];
    }];
}

- (BOOL)insertContact:(Contact *)contact {
    __block BOOL success = NO;
    [_queue inDatabase:^(FMDatabase *db) {
        NSString *sql = @"INSERT INTO contact (name, phone, avatar) VALUES (?, ?, ?)";
        success = [db executeUpdate:sql, contact.name, contact.phone,
                   contact.avatarPath ?: [NSNull null]];
    }];
    return success;
}

- (BOOL)deleteContact:(NSInteger)contactId {
    __block BOOL success = NO;
    [_queue inDatabase:^(FMDatabase *db) {
        NSString *sql = @"DELETE FROM contact WHERE id = ?";
        success = [db executeUpdate:sql, @(contactId)];
    }];
    return success;
}

- (BOOL)updateContact:(Contact *)contact {
    __block BOOL success = NO;
    [_queue inDatabase:^(FMDatabase *db) {
        NSString *sql = @"UPDATE contact SET name = ?, phone = ?, avatar = ? WHERE id = ?";
        success = [db executeUpdate:sql, contact.name, contact.phone,
                   contact.avatarPath ?: [NSNull null], @(contact.contactId)];
    }];
    return success;
}

- (NSArray<Contact *> *)allContacts {
    NSMutableArray *result = [NSMutableArray array];
    [_queue inDatabase:^(FMDatabase *db) {
        FMResultSet *rs = [db executeQuery:@"SELECT * FROM contact"];
        while ([rs next]) {
            Contact *c = [[Contact alloc] init];
            c.contactId  = [rs intForColumn:@"id"];
            c.name       = [rs stringForColumn:@"name"];
            c.phone      = [rs stringForColumn:@"phone"];
            c.avatarPath = [rs stringForColumn:@"avatar"];
            [result addObject:c];
        }
        [rs close];
    }];
    return result;
}

- (BOOL)isNameExist:(NSString *)name {
    __block BOOL exist = NO;
    [_queue inDatabase:^(FMDatabase *db) {
        FMResultSet *rs = [db executeQuery:@"SELECT COUNT(*) FROM contact WHERE name = ?", name];
        if ([rs next]) {
            exist = [rs intForColumnIndex:0] > 0;
        }
        [rs close];
    }];
    return exist;
}

@end
```

```objc
#import "ContactListViewController.h"
#import "ContactEditViewController.h"
#import "DBManager.h"
#import "Contact.h"
#import <Masonry/Masonry.h>

@interface ContactListViewController () <UITableViewDataSource, UITableViewDelegate>
@property (nonatomic, strong) UITableView *tableView;
@property (nonatomic, strong) NSMutableArray *contacts;
@end

@implementation ContactListViewController

- (void)viewDidLoad {
    [super viewDidLoad];
    self.title = @"通讯录";

    // 右上角：新增按钮
    self.navigationItem.rightBarButtonItem =
        [[UIBarButtonItem alloc] initWithBarButtonSystemItem:UIBarButtonSystemItemAdd
                                                      target:self
                                                      action:@selector(addContact)];

    self.tableView = [[UITableView alloc] initWithFrame:CGRectZero style:UITableViewStylePlain];
    self.tableView.dataSource = self;
    self.tableView.delegate = self;
    [self.view addSubview:self.tableView];

    [self.tableView mas_makeConstraints:^(MASConstraintMaker *make) {
        make.edges.equalTo(self.view);
    }];
}

// 每次出现重新读数据库
- (void)viewWillAppear:(BOOL)animated {
    [super viewWillAppear:animated];
    self.contacts = [[[DBManager sharedManager] allContacts] mutableCopy];
    [self.tableView reloadData];
}

// 右上角 + ：新增
- (void)addContact {
    ContactEditViewController *editVC = [[ContactEditViewController alloc] init];
    [self.navigationController pushViewController:editVC animated:YES];
}

#pragma mark - TableView 数据源

- (NSInteger)tableView:(UITableView *)tableView numberOfRowsInSection:(NSInteger)section {
    return self.contacts.count;
}

- (UITableViewCell *)tableView:(UITableView *)tableView cellForRowAtIndexPath:(NSIndexPath *)indexPath {
    static NSString *cellId = @"cell";
    UITableViewCell *cell = [tableView dequeueReusableCellWithIdentifier:cellId];
    if (!cell) {
        cell = [[UITableViewCell alloc] initWithStyle:UITableViewCellStyleSubtitle reuseIdentifier:cellId];
    }

    Contact *c = self.contacts[indexPath.row];

    if (c.avatarPath) {
        cell.imageView.image = [UIImage imageWithContentsOfFile:c.avatarPath];
    } else {
        cell.imageView.image = [UIImage imageNamed:@"默认头像"];
    }
    cell.textLabel.text = c.name;
    cell.detailTextLabel.text = c.phone;

    return cell;
}

#pragma mark - 点击编辑 + 左滑删除

// 点击某一行 → 编辑联系人
- (void)tableView:(UITableView *)tableView didSelectRowAtIndexPath:(NSIndexPath *)indexPath {
    Contact *c = self.contacts[indexPath.row];
    ContactEditViewController *editVC = [[ContactEditViewController alloc] init];
    editVC.editingContact = c;   // 传数据 = 编辑模式
    [self.navigationController pushViewController:editVC animated:YES];
}

// 左滑删除
- (void)tableView:(UITableView *)tableView
    commitEditingStyle:(UITableViewCellEditingStyle)editingStyle
     forRowAtIndexPath:(NSIndexPath *)indexPath {
    if (editingStyle == UITableViewCellEditingStyleDelete) {
        Contact *c = self.contacts[indexPath.row];
        [[DBManager sharedManager] deleteContact:c.contactId];          // 删数据库
        [self.contacts removeObjectAtIndex:indexPath.row];              // 删数组
        [tableView deleteRowsAtIndexPaths:@[indexPath] withRowAnimation:UITableViewRowAnimationFade];
    }
}

@end
```

## ContactEditViewController.m

```objc
#import "ContactEditViewController.h"
#import "DBManager.h"
#import <Masonry/Masonry.h>

@interface ContactEditViewController () <UIImagePickerControllerDelegate, UINavigationControllerDelegate>
@property (nonatomic, strong) UIImageView *avatarView;
@property (nonatomic, strong) UITextField *nameField;
@property (nonatomic, strong) UITextField *phoneField;
@property (nonatomic, copy)   NSString *selectedAvatarPath;
@end

@implementation ContactEditViewController

- (void)viewDidLoad {
    [super viewDidLoad];
    self.view.backgroundColor = [UIColor whiteColor];
    self.title = self.editingContact ? @"编辑联系人" : @"添加联系人";

    [self setupUI];

    // 编辑模式回填数据
    if (self.editingContact) {
        self.nameField.text = self.editingContact.name;
        self.phoneField.text = self.editingContact.phone;
        self.selectedAvatarPath = self.editingContact.avatarPath;
        self.avatarView.image = self.editingContact.avatarPath
            ? [UIImage imageWithContentsOfFile:self.editingContact.avatarPath]
            : [UIImage imageNamed:@"默认头像"];
    } else {
        self.avatarView.image = [UIImage imageNamed:@"默认头像"];
    }
}

- (void)setupUI {
    // 头像（点击选相册）
    self.avatarView = [[UIImageView alloc] init];
    self.avatarView.contentMode = UIViewContentModeScaleAspectFit;
    self.avatarView.userInteractionEnabled = YES;
    [self.avatarView addGestureRecognizer:
        [[UITapGestureRecognizer alloc] initWithTarget:self action:@selector(chooseAvatar)]];
    [self.view addSubview:self.avatarView];

    // 姓名
    self.nameField = [[UITextField alloc] init];
    self.nameField.placeholder = @"请输入姓名";
    self.nameField.borderStyle = UITextBorderStyleRoundedRect;
    [self.view addSubview:self.nameField];

    // 电话
    self.phoneField = [[UITextField alloc] init];
    self.phoneField.placeholder = @"请输入11位电话号码";
    self.phoneField.keyboardType = UIKeyboardTypeNumberPad;
    self.phoneField.borderStyle = UITextBorderStyleRoundedRect;
    [self.view addSubview:self.phoneField];

    // 保存按钮
    UIButton *saveBtn = [UIButton buttonWithType:UIButtonTypeSystem];
    [saveBtn setTitle:@"保存" forState:UIControlStateNormal];
    [saveBtn addTarget:self action:@selector(saveButtonClicked)
      forControlEvents:UIControlEventTouchUpInside];
    [self.view addSubview:saveBtn];

    // 约束
    [self.avatarView mas_makeConstraints:^(MASConstraintMaker *make) {
        make.top.equalTo(self.view.mas_safeAreaLayoutGuideTop).offset(40);
        make.centerX.equalTo(self.view);
        make.width.height.mas_equalTo(80);
    }];

    [self.nameField mas_makeConstraints:^(MASConstraintMaker *make) {
        make.top.equalTo(self.avatarView.mas_bottom).offset(30);
        make.left.equalTo(self.view).offset(30);
        make.right.equalTo(self.view).offset(-30);
        make.height.mas_equalTo(44);
    }];

    [self.phoneField mas_makeConstraints:^(MASConstraintMaker *make) {
        make.top.equalTo(self.nameField.mas_bottom).offset(20);
        make.left.equalTo(self.view).offset(30);
        make.right.equalTo(self.view).offset(-30);
        make.height.mas_equalTo(44);
    }];

    [saveBtn mas_makeConstraints:^(MASConstraintMaker *make) {
        make.top.equalTo(self.phoneField.mas_bottom).offset(40);
        make.centerX.equalTo(self.view);
        make.width.mas_equalTo(120);
        make.height.mas_equalTo(44);
    }];
}

#pragma mark - 头像（相册）

- (void)chooseAvatar {
    UIImagePickerController *picker = [[UIImagePickerController alloc] init];
    picker.sourceType = UIImagePickerControllerSourceTypePhotoLibrary;
    picker.delegate = self;
    [self presentViewController:picker animated:YES completion:nil];
}

- (void)imagePickerController:(UIImagePickerController *)picker
    didFinishPickingMediaWithInfo:(NSDictionary<UIImagePickerControllerInfoKey, id> *)info {
    UIImage *image = info[UIImagePickerControllerOriginalImage];
    self.avatarView.image = image;
    self.selectedAvatarPath = [self saveImage:image];
    [picker dismissViewControllerAnimated:YES completion:nil];
}

// 图片存沙盒，返回路径
- (NSString *)saveImage:(UIImage *)image {
    NSString *docPath = [NSSearchPathForDirectoriesInDomains(NSDocumentDirectory, NSUserDomainMask, YES) firstObject];
    NSString *fileName = [NSString stringWithFormat:@"avatar_%ld.png", (long)[[NSDate date] timeIntervalSince1970]];
    NSString *filePath = [docPath stringByAppendingPathComponent:fileName];
    [UIImagePNGRepresentation(image) writeToFile:filePath atomically:YES];
    return filePath;
}

#pragma mark - 保存（含校验 + 判重）

- (void)saveButtonClicked {
    NSString *name = self.nameField.text;
    NSString *phone = self.phoneField.text;

    // 电话 11 位
    if (phone.length != 11) {
        NSLog(@"电话必须是11位");
        return;
    }

    // 名字判重
    if ([[DBManager sharedManager] isNameExist:name]) {
        NSLog(@"姓名已存在");
        return;
    }

    // 编辑 or 新增
    if (self.editingContact) {
        self.editingContact.name = name;
        self.editingContact.phone = phone;
        self.editingContact.avatarPath = self.selectedAvatarPath;
        [[DBManager sharedManager] updateContact:self.editingContact];
    } else {
        Contact *c = [[Contact alloc] init];
        c.name = name;
        c.phone = phone;
        c.avatarPath = self.selectedAvatarPath;
        [[DBManager sharedManager] insertContact:c];
    }

    [self.navigationController popViewControllerAnimated:YES];
}

@end
```

## ⑤ SceneDelegate.m（设置根控制器）

```objc
- (void)scene:(UIScene *)scene willConnectToSession:(UISceneSession *)session
         options:(UISceneConnectionOptions *)connectionOptions {

    UIWindowScene *windowScene = (UIWindowScene *)scene;
    self.window = [[UIWindow alloc] initWithWindowScene:windowScene];

    ContactListViewController *listVC = [[ContactListViewController alloc] init];
    UINavigationController *nav = [[UINavigationController alloc] initWithRootViewController:listVC];
    self.window.rootViewController = nav;

    [self.window makeKeyAndVisible];
}
```
